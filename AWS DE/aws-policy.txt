################### Devops Admin level Policy ###########################

# Define the IAM policy
resource "aws_iam_policy" "ti-devopsadmin_policy" {
  name        = "ti-devopsadmin"
  description = "Policy for Tidevops administrators"
  policy      = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Effect   = "Allow"
        Action   = [
          "ec2:*",
          "s3:*",
          "rds:*",
          "logs:*",
          "elasticloadbalancing:*",
          "cloudformation:*",
          "cloudwatch:*",
          "lambda:*",
          "autoscaling:*",
          "sns:*",
          "access-analyzer:ListPolicyGenerations",
          "iam:*",
          "glue:*",
          "athena:*",
          "access-analyzer:ValidatePolicy",
          "logs:FilterLogEvents",
          "logs:DescribeLogGroups",
          "logs:PutLogEvents"
        ]
        Resource = "*"
      }
    ]
  })
}


################### Instructor level Policy for S3 ###########################

  # Check if the policy already exists
  data "aws_iam_policy" "ti-p-instructor-s3-policy" {
  name = "ti-p-instructor-s3-policy"
 }

# Create the policy if it doesn't exist
resource "aws_iam_policy" "ti-p-instructor-s3-policy" {
  name = "ti-p-instructor-s3-policy"
  # Only create the policy if the data source doesn't have any results
  lifecycle {
    create_before_destroy = true
  }

  policy = <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": [
        "s3:GetObject",
        "s3:PutObject",
        "s3:DeleteObject",
        "s3:ListBucket"
      ],
      "Resource": [
        "arn:aws:s3:::ti-p*"        
      ]
    }
  ]
}
EOF
}

  name        = "lambda-policy"
  description = "Policy granting permissions to create and execute Lambda functions"
  policy      = jsonencode({
    Version = "2012-10-17",
    Statement = [
      {
        Effect   = "Allow",
        Action   = [
          "lambda:CreateFunction",
          "lambda:InvokeFunction",
          "lambda:InvokeAsync",
          "lambda:UpdateFunctionCode",
          "lambda:UpdateFunctionConfiguration",
          "lambda:DeleteFunction",
          "lambda:GetFunction"
        ],
        Resource = "*"
      }
    ]
  })
}

  name       = "lambda-policy-attachment"
  users      = [aws_iam_user.lambda_user.name]
  policy_arn = aws_iam_policy.lambda_policy.arn
}
