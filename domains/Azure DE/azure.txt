# How to run (Azure RBAC version of your AWS IAM policies)
# 1. Go to the folder you want to execute the run
# 2. terraform init
# 3. terraform plan
# 4. terraform apply
#
# Notes:
# - Azure doesn't have "IAM Policies" like AWS; it uses Azure RBAC roles assigned at a scope.
# - This file creates Entra ID (Azure AD) Groups and assigns built-in roles to them.
# - Storage permission is granted using "Storage Blob Data *" roles (data-plane).
# - Creating Entra users/groups requires directory permissions; if your tenant restricts it,
#   keep the groups and use existing users, or have an Entra admin run it.
#
# Mapping (roughly):
# - AWS "AdministratorAccess / devopsadmin"  -> Azure RBAC "Owner" at subscription scope
# - AWS S3 bucket policies                   -> Azure RBAC on Storage Accounts/Containers
# - AWS Glue                                 -> Azure Data Factory (ADF) roles
# - AWS Athena read-only                      -> Azure "Reader" on your analytics scope (Synapse/ADX/RG)
# - AWS Lambda user/policy                    -> Azure Functions Contributor (or Contributor) on RG
#
# You can copy this into a main.tf and adjust variables as needed.

terraform {
  required_version = ">= 1.5.0"

  required_providers {
    azurerm = {
      source  = "hashicorp/azurerm"
      version = ">= 3.100.0"
    }
    azuread = {
      source  = "hashicorp/azuread"
      version = ">= 2.50.0"
    }
    random = {
      source  = "hashicorp/random"
      version = ">= 3.6.0"
    }
  }
}

# ----------------------------
# Providers
# ----------------------------
provider "azurerm" {
  features {}

  # Prefer environment variables for auth:
  # ARM_SUBSCRIPTION_ID, ARM_TENANT_ID, ARM_CLIENT_ID, ARM_CLIENT_SECRET
  subscription_id = var.subscription_id
  tenant_id       = var.tenant_id
}

provider "azuread" {
  tenant_id = var.tenant_id
}

data "azurerm_client_config" "current" {}
data "azurerm_subscription" "current" {}

# ----------------------------
# Variables
# ----------------------------
variable "subscription_id" {
  type        = string
  description = "Azure Subscription ID"
}

variable "tenant_id" {
  type        = string
  description = "Azure Tenant (Directory) ID"
}

variable "location" {
  type        = string
  default     = "eastus"
  description = "Azure region"
}

variable "resource_group_name" {
  type        = string
  default     = "ti-platform-rg"
  description = "Resource group for training resources"
}

# If you already have an analytics resource (Synapse/ADX/etc), set its scope id here.
# Example: /subscriptions/<sub>/resourceGroups/<rg>/providers/Microsoft.Synapse/workspaces/<ws>
# If empty, this module will default to the Resource Group scope.
variable "analytics_scope_id" {
  type        = string
  default     = ""
  description = "Scope ID for analytics read-only group (Synapse/ADX/RG)."
}

# ----------------------------
# Resource Group
# ----------------------------
resource "azurerm_resource_group" "ti_rg" {
  name     = var.resource_group_name
  location = var.location
}

# ----------------------------
# Storage Accounts (Azure equivalent of your S3 buckets)
# NOTE: Azure storage account names must be globally unique, 3-24 chars, lowercase/numbers only.
# We'll add a random suffix.
# ----------------------------
resource "random_string" "suffix" {
  length  = 6
  upper   = false
  lower   = true
  numeric = true
  special = false
}

# "ti-p-students" bucket analogue
resource "azurerm_storage_account" "students_sa" {
  name                     = "tipstudents${random_string.suffix.result}"
  resource_group_name      = azurerm_resource_group.ti_rg.name
  location                 = azurerm_resource_group.ti_rg.location
  account_tier             = "Standard"
  account_replication_type = "LRS"
  allow_nested_items_to_be_public = false
}

# "ti-p-etl-data" bucket analogue
resource "azurerm_storage_account" "etl_sa" {
  name                     = "tipetldata${random_string.suffix.result}"
  resource_group_name      = azurerm_resource_group.ti_rg.name
  location                 = azurerm_resource_group.ti_rg.location
  account_tier             = "Standard"
  account_replication_type = "LRS"
  allow_nested_items_to_be_public = false
}

# "ti-p*" instructor bucket analogue (a shared training storage account)
resource "azurerm_storage_account" "instructor_sa" {
  name                     = "tipinstructor${random_string.suffix.result}"
  resource_group_name      = azurerm_resource_group.ti_rg.name
  location                 = azurerm_resource_group.ti_rg.location
  account_tier             = "Standard"
  account_replication_type = "LRS"
  allow_nested_items_to_be_public = false
}

# Optional: create a few containers to mirror buckets/folders
resource "azurerm_storage_container" "students_container" {
  name                  = "students"
  storage_account_name  = azurerm_storage_account.students_sa.name
  container_access_type = "private"
}

resource "azurerm_storage_container" "etl_container" {
  name                  = "etl-data"
  storage_account_name  = azurerm_storage_account.etl_sa.name
  container_access_type = "private"
}

resource "azurerm_storage_container" "instructor_container" {
  name                  = "instructor"
  storage_account_name  = azurerm_storage_account.instructor_sa.name
  container_access_type = "private"
}

# ----------------------------
# Data Factory (Azure equivalent of AWS Glue for this mapping)
# ----------------------------
resource "azurerm_data_factory" "ti_adf" {
  name                = "ti-adf-${random_string.suffix.result}"
  location            = azurerm_resource_group.ti_rg.location
  resource_group_name = azurerm_resource_group.ti_rg.name
}

# ----------------------------
# Entra ID (Azure AD) Groups (equivalent of AWS IAM Groups)
# ----------------------------
resource "azuread_group" "ti_devopsadmin_group" {
  display_name     = "ti-devopsadmin"
  security_enabled = true
}

resource "azuread_group" "ti_instructor_s3_group" {
  display_name     = "ti-p-instructor-storage"
  security_enabled = true
}

resource "azuread_group" "ti_developer_s3_group" {
  display_name     = "ti-p-developer-storage"
  security_enabled = true
}

resource "azuread_group" "ti_instructor_adf_group" {
  display_name     = "ti-p-instructor-adf"
  security_enabled = true
}

resource "azuread_group" "ti_developer_adf_group" {
  display_name     = "ti-p-developer-adf"
  security_enabled = true
}

resource "azuread_group" "ti_analytics_readonly_group" {
  display_name     = "athena-readonly-group"
  security_enabled = true
}

resource "azuread_group" "ti_viz_readonly_group" {
  display_name     = "ti-storage-viz-readonly"
  security_enabled = true
}

resource "azuread_group" "ti_viz_readwrite_group" {
  display_name     = "ti-storage-viz-readwrite"
  security_enabled = true
}

# ----------------------------
# RBAC Role Assignments (equivalent of attaching IAM Policies to groups)
# ----------------------------

# 1) DevOps Admin level (full admin) => Owner at subscription scope
resource "azurerm_role_assignment" "ti_devopsadmin_owner" {
  scope                = "/subscriptions/${var.subscription_id}"
  role_definition_name = "Owner"
  principal_id         = azuread_group.ti_devopsadmin_group.object_id
}

# 2) Instructor level Storage (Get/Put/Delete/List) => Storage Blob Data Contributor
resource "azurerm_role_assignment" "ti_instructor_storage_contributor" {
  scope                = azurerm_storage_account.instructor_sa.id
  role_definition_name = "Storage Blob Data Contributor"
  principal_id         = azuread_group.ti_instructor_s3_group.object_id
}

# 3) ETL Developer Storage (Get/Put/List) => Storage Blob Data Contributor on ETL storage
resource "azurerm_role_assignment" "ti_developer_storage_contributor" {
  scope                = azurerm_storage_account.etl_sa.id
  role_definition_name = "Storage Blob Data Contributor"
  principal_id         = azuread_group.ti_developer_s3_group.object_id
}

# 4) Instructor ADF (Glue:*) => Data Factory Contributor
resource "azurerm_role_assignment" "ti_instructor_adf_contributor" {
  scope                = azurerm_data_factory.ti_adf.id
  role_definition_name = "Data Factory Contributor"
  principal_id         = azuread_group.ti_instructor_adf_group.object_id
}

# 5) ETL Developer ADF (limited) => Data Factory Reader (read-only)
resource "azurerm_role_assignment" "ti_developer_adf_reader" {
  scope                = azurerm_data_factory.ti_adf.id
  role_definition_name = "Data Factory Reader"
  principal_id         = azuread_group.ti_developer_adf_group.object_id
}

# 6) "Athena read-only" analogue:
# Use Reader on an analytics scope (Synapse/ADX/etc). If not provided, Reader on the RG.
locals {
  analytics_scope = var.analytics_scope_id != "" ? var.analytics_scope_id : azurerm_resource_group.ti_rg.id
}

resource "azurerm_role_assignment" "athena_readonly_reader" {
  scope                = local.analytics_scope
  role_definition_name = "Reader"
  principal_id         = azuread_group.ti_analytics_readonly_group.object_id
}

# 7) Visualization Storage read-only => Storage Blob Data Reader on all storage accounts
resource "azurerm_role_assignment" "viz_storage_readonly_students" {
  scope                = azurerm_storage_account.students_sa.id
  role_definition_name = "Storage Blob Data Reader"
  principal_id         = azuread_group.ti_viz_readonly_group.object_id
}

resource "azurerm_role_assignment" "viz_storage_readonly_etl" {
  scope                = azurerm_storage_account.etl_sa.id
  role_definition_name = "Storage Blob Data Reader"
  principal_id         = azuread_group.ti_viz_readonly_group.object_id
}

resource "azurerm_role_assignment" "viz_storage_readonly_instructor" {
  scope                = azurerm_storage_account.instructor_sa.id
  role_definition_name = "Storage Blob Data Reader"
  principal_id         = azuread_group.ti_viz_readonly_group.object_id
}

# 8) Visualization Storage read-write => Contributor on students SA
resource "azurerm_role_assignment" "viz_storage_readwrite_students" {
  scope                = azurerm_storage_account.students_sa.id
  role_definition_name = "Storage Blob Data Contributor"
  principal_id         = azuread_group.ti_viz_readwrite_group.object_id
}

# ----------------------------
# Lambda user analogue -> Azure Functions user (optional)
# ----------------------------
variable "create_function_user" {
  type        = bool
  default     = false
  description = "Set true if you want Terraform to create a demo function user (requires Entra permissions)."
}

variable "function_user_upn" {
  type        = string
  default     = "lambda-user@yourtenant.onmicrosoft.com"
  description = "UPN for the function user (only used if create_function_user=true)."
}

variable "function_user_password" {
  type        = string
  default     = "ChangeM3Now!ChangeM3Now!"
  description = "Temporary password (only used if create_function_user=true)."
  sensitive   = true
}

resource "azuread_user" "function_user" {
  count               = var.create_function_user ? 1 : 0
  user_principal_name = var.function_user_upn
  display_name        = "lambda-user"
  password            = var.function_user_password
}

resource "azurerm_role_assignment" "function_user_rg" {
  count                = var.create_function_user ? 1 : 0
  scope                = azurerm_resource_group.ti_rg.id
  role_definition_name = "Azure Functions Contributor"
  principal_id         = azuread_user.function_user[0].object_id
}

# ----------------------------
# Outputs
# ----------------------------
output "resource_group_id" {
  value = azurerm_resource_group.ti_rg.id
}

output "storage_accounts" {
  value = {
    students_sa   = azurerm_storage_account.students_sa.name
    etl_sa        = azurerm_storage_account.etl_sa.name
    instructor_sa = azurerm_storage_account.instructor_sa.name
  }
}

output "group_object_ids" {
  value = {
    ti_devopsadmin                 = azuread_group.ti_devopsadmin_group.object_id
    ti_instructor_storage          = azuread_group.ti_instructor_s3_group.object_id
    ti_developer_storage           = azuread_group.ti_developer_s3_group.object_id
    ti_instructor_adf              = azuread_group.ti_instructor_adf_group.object_id
    ti_developer_adf               = azuread_group.ti_developer_adf_group.object_id
    athena_readonly                = azuread_group.ti_analytics_readonly_group.object_id
    ti_viz_storage_readonly        = azuread_group.ti_viz_readonly_group.object_id
    ti_viz_storage_readwrite       = azuread_group.ti_viz_readwrite_group.object_id
  }
}
